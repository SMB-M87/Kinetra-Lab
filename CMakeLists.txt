cmake_minimum_required(VERSION 3.16)
project(KinetraLab LANGUAGES C CXX)
include(CTest)
option(KINETRA_LAB_BUILD_RUNTIME "Build graphical applications and launcher" ON)
option(KINETRA_TEST_GRAPHICS "Enable tests requiring an OpenGL context" OFF)
set(KINETRA_SOURCE_DIR "" CACHE PATH "Kinetra source checkout; empty uses installed package, then sibling")

# Prefer an explicitly selected source checkout, otherwise an installed package.
if(NOT KINETRA_SOURCE_DIR)
    find_package(Kinetra 0.1 CONFIG QUIET)
    if(NOT Kinetra_FOUND AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../Kinetra/CMakeLists.txt")
        set(KINETRA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../Kinetra")
    endif()
endif()
if(KINETRA_SOURCE_DIR)
    set(KINETRA_BUILD_RUNTIME ${KINETRA_LAB_BUILD_RUNTIME} CACHE BOOL "" FORCE)
    # Keep core tests independent of Lab's CTest suites.
    set(lab_testing ${BUILD_TESTING})
    set(BUILD_TESTING OFF)
    add_subdirectory("${KINETRA_SOURCE_DIR}" kinetra-core)
    set(BUILD_TESTING ${lab_testing})
elseif(NOT Kinetra_FOUND)
    message(FATAL_ERROR "Provide KINETRA_SOURCE_DIR, a sibling Kinetra checkout, or CMAKE_PREFIX_PATH to an installed Kinetra")
endif()
if(KINETRA_LAB_BUILD_RUNTIME AND NOT TARGET Kinetra::Runtime)
    message(FATAL_ERROR "The selected Kinetra package needs KINETRA_BUILD_RUNTIME=ON")
endif()

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
# Application CMake uses this local switch for its existing presentation targets.
set(KINETRA_BUILD_RUNTIME ${KINETRA_LAB_BUILD_RUNTIME})
add_subdirectory(demo)
add_subdirectory(dotio)
if(KINETRA_LAB_BUILD_RUNTIME)
    add_subdirectory(launcher)
endif()
